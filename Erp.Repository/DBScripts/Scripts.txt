-- =============================================
-- Stored Procedure: sp_GetProductionBatchList
-- Returns production batch list with supplier name,
-- supervisor name, and total quantity for the UI grid.
-- =============================================
IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'sp_GetProductionBatchList' AND type = 'P')
    DROP PROCEDURE [dbo].[sp_GetProductionBatchList];
GO

  
create  PROCEDURE [dbo].[sp_GetProductionBatchList]  
    @SearchText NVARCHAR(255) = NULL,  
    @FromDate   DATETIME2     = NULL,  
    @ToDate     DATETIME2     = NULL  
AS  
BEGIN  
    SET NOCOUNT ON;  
  
    SELECT  
        pb.[Id],  
        pb.[Production_Batch_Id],  
        pb.[Production_Batch_Number],    
        pb.[Start_Date],  
        pb.[End_Date]   ,
		isnull(sum(pd.Input_Quantity),0) as total_Input_Qty
    FROM [dbo].[Production_Batch_Master] pb   
	inner join [Production_Batch_Details] pd on pd.Production_Batch_Id=pb.Production_Batch_Id 
    WHERE pb.[DeletedAt] IS NULL  
      AND (@SearchText IS NULL OR pb.[Production_Batch_Number] LIKE '%' + @SearchText + '%')  
      AND (@FromDate IS NULL OR pb.[Start_Date] >= @FromDate)  
      AND (@ToDate   IS NULL OR pb.[Start_Date] <= @ToDate)  
	  group by    pb.[Id],  
        pb.[Production_Batch_Id],  
        pb.[Production_Batch_Number],    
        pb.[Start_Date],  
        pb.[End_Date] 
    ORDER BY pb.[Id] DESC;  
END;  
GO

-- =============================================
-- Stored Procedure: sp_GetOpenJobReceipts
-- Returns job receipt list joined with items
-- for the UI grid, filtered by date range.
-- =============================================
IF EXISTS (SELECT 1 FROM sys.objects WHERE name = 'sp_GetOpenJobReceipts' AND type = 'P')
    DROP PROCEDURE [dbo].[sp_GetOpenJobReceipts];
GO

create PROCEDURE [dbo].[sp_GetOpenJobReceipts]
    @FromDate DATETIME2 = NULL,
    @ToDate   DATETIME2 = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        jrm.[Job_Receipt_Id],
        jrm.[Job_Receipt_Number],
        jrm.[Job_Receipt_Date],
        ISNULL(s.[Supplier_Name], 'N/A') AS [Supplier_Name],
        ISNULL(
            jrm.[Total_Quantity],
            CAST(0 AS DECIMAL(18,3))
        ) AS [Total_Qty],
        ISNULL(
            jrm.[Total_Amount],
            CAST(0 AS DECIMAL(18,3))
        ) AS [Total_Amount],
        CAST(0 AS BIT) AS [Is_Selected]

    FROM [dbo].[Job_Receipt_Master] jrm

    LEFT JOIN [dbo].[Supplier] s
        ON s.[Supplier_Id] = jrm.[Supplier_Id]

    WHERE jrm.[DeletedAt] IS NULL

     -- AND (
   ----         @FromDate IS NULL
            OR jrm.[Job_Receipt_Date] >= @FromDate
     --     )

    ----  AND (
        --    @ToDate IS NULL
       --     OR jrm.[Job_Receipt_Date] < DATEADD(DAY, 1, CAST(@ToDate AS DATE))
        --  )
		  and  Job_Receipt_Id not in (select Job_Receipt_Master_Id from Production_Batch_Details)

    ORDER BY jrm.[Id] DESC;
END;

        ) AS [Total_Amount],    
    
        -- Indicates whether Job Receipt is already added    
        CAST(1 AS BIT) AS [Is_Selected],    
    
        -- Production Batch details    
        pd.[Id] AS [Production_Batch_Detail_Id],    
        pd.[Production_Batch_Id],    
        pd.[Job_Receipt_Master_Id]    
    
    FROM dbo.[Job_Receipt_Master] AS jm    
    
    LEFT JOIN dbo.[Supplier] AS s    
        ON s.[Supplier_Id] = jm.[Supplier_Id]    
    
    INNER JOIN dbo.[Production_Batch_Details] AS pd    
        ON pd.[Job_Receipt_Master_Id] = jm.[Job_Receipt_Id]    
        AND pd.[Production_Batch_Id] = @Production_Batch_Id    
    
    WHERE    
        jm.[DeletedAt] IS NULL    
    
        --AND (    
        --    @FromDate IS NULL    
        --    OR jm.[Job_Receipt_Date] >= @FromDate    
        --)    
    
        --AND (    
        --    @ToDate IS NULL    
        --    OR jm.[Job_Receipt_Date] < DATEADD(DAY, 1, CAST(@ToDate AS DATE))    
        --)    
    
    UNION    
    
    SELECT    
        jrm.[Job_Receipt_Id],    
        jrm.[Job_Receipt_Number],    
        jrm.[Job_Receipt_Date],    
    
        ISNULL(    
            s.[Supplier_Name],    
            'N/A'    
        ) AS [Supplier_Name],    
    
        ISNULL(    
            jrm.[Total_Quantity],    
            CAST(0 AS DECIMAL(18,3))    
        ) AS [Total_Qty],    
    
        ISNULL(    
            jrm.[Total_Amount],    
            CAST(0 AS DECIMAL(18,3))    
        ) AS [Total_Amount],    
    
        CAST(0 AS BIT) AS [Is_Selected],    
    
        -- Production_Batch_Details.Id is INT    
        CAST(0 AS INT) AS [Production_Batch_Detail_Id],    
    
        -- These are UNIQUEIDENTIFIER    
        CAST(    
            '00000000-0000-0000-0000-000000000000'    
            AS UNIQUEIDENTIFIER    
        ) AS [Production_Batch_Id],    
    
        CAST(    
            '00000000-0000-0000-0000-000000000000'    
            AS UNIQUEIDENTIFIER    
        ) AS [Job_Receipt_Master_Id]    
    
    FROM dbo.[Job_Receipt_Master] AS jrm    
    
    LEFT JOIN dbo.[Supplier] AS s    
        ON s.[Supplier_Id] = jrm.[Supplier_Id]    
    
    WHERE    
        jrm.[DeletedAt] IS NULL    
    
        --AND (    
        --    @FromDate IS NULL    
        --    OR jrm.[Job_Receipt_Date] >= @FromDate    
        --)    
    
        --AND (    
        --    @ToDate IS NULL    
        --    OR jrm.[Job_Receipt_Date] < DATEADD(DAY, 1, CAST(@ToDate AS DATE))    
        --)    
    
        AND NOT EXISTS    
        (    
            SELECT 1    
            FROM dbo.[Production_Batch_Details] AS pbd    
            WHERE pbd.[Job_Receipt_Master_Id] = jrm.[Job_Receipt_Id]    
        )    
    
    ORDER BY    
        [Job_Receipt_Date] DESC;    
    

  
END;

CREATE PROCEDURE [dbo].sp_GetProductionBatchCostList     
    @SearchText NVARCHAR(255) = NULL,      
    @FromDate   DATETIME2     = NULL,      
    @ToDate     DATETIME2     = NULL      
AS      
BEGIN      
    SET NOCOUNT ON;      
      
    SELECT      
        pb.[Id],      
        pb.[Production_Batch_Id],      
        pb.[Production_Batch_Number],        
        pb.[Start_Date],      
        pb.[End_Date]   ,    
  isnull(sum(pd.Input_Quantity),0) as total_Input_Qty    
    FROM [dbo].[Production_Batch_Master] pb       
 inner join [Production_Batch_Details] pd on pd.Production_Batch_Id=pb.Production_Batch_Id     
    WHERE pb.[DeletedAt] IS NULL      
      AND (@SearchText IS NULL OR pb.[Production_Batch_Number] LIKE '%' + @SearchText + '%')      
      AND (@FromDate IS NULL OR pb.[Start_Date] >= @FromDate)      
      AND (@ToDate   IS NULL OR pb.[Start_Date] <= @ToDate)      
	  and pb.Production_Batch_Id not in (select Production_Batch_Id from Production_Cost_Master)
   group by    pb.[Id],      
        pb.[Production_Batch_Id],      
        pb.[Production_Batch_Number],        
        pb.[Start_Date],      
        pb.[End_Date]     
    ORDER BY pb.[Id] DESC;      
END;

CREATE   PROCEDURE dbo.GetMonthlyReport  
(  
    @SearchText NVARCHAR(100) = NULL,  
    @FromDate   DATETIME2 = NULL,  
    @ToDate     DATETIME2 = NULL  
)  
AS  
BEGIN  
    SET NOCOUNT ON;  
  
    ;WITH BatchData AS  
    (  
        SELECT  
            PB.Id AS BatchId,  
            PB.Production_Batch_Id,  
            PB.Production_Batch_Number,  
            PB.Start_Date  
        FROM Production_Batch_Master PB  
        WHERE PB.DeletedAt IS NULL  
          AND  
          (  
              NULLIF(LTRIM(RTRIM(@SearchText)), '') IS NULL  
              OR PB.Production_Batch_Number LIKE '%' + LTRIM(RTRIM(@SearchText)) + '%'  
          )  
          AND  
          (  
              @FromDate IS NULL  
              OR PB.Start_Date >= @FromDate  
          )  
          AND  
          (  
              @ToDate IS NULL  
              OR PB.Start_Date < DATEADD(DAY, 1, CAST(@ToDate AS DATE))  
          )  
    ),  
  
    -- Raw material quantity  
    RawMaterialQtyData AS  
    (  
        SELECT  
            PBD.Production_Batch_Id,  
            SUM(ISNULL(PBD.Input_Quantity, 0)) AS RawMaterialQty  
        FROM Production_Batch_Details PBD  
        INNER JOIN BatchData B  
            ON B.Production_Batch_Id = PBD.Production_Batch_Id  
        GROUP BY  
            PBD.Production_Batch_Id  
    ),  
  
    -- Material cost  
    -- DISTINCT is important to match the original C# ToHashSet() logic  
    MaterialCostData AS  
    (  
        SELECT  
            X.Production_Batch_Id,  
            SUM(ISNULL(JRM.Total_Amount, 0)) AS MaterialCost  
        FROM  
        (  
            SELECT DISTINCT  
                PBD.Production_Batch_Id,  
                PBD.Job_Receipt_Master_Id  
            FROM Production_Batch_Details PBD  
            INNER JOIN BatchData B  
                ON B.Production_Batch_Id = PBD.Production_Batch_Id  
        ) X  
        INNER JOIN Job_Receipt_Master JRM  
            ON JRM.Job_Receipt_Id = X.Job_Receipt_Master_Id  
           AND JRM.DeletedAt IS NULL  
        GROUP BY  
            X.Production_Batch_Id  
    ),  
  
    -- Production and wastage  
    ProductionData AS  
    (  
        SELECT  
            D.Production_Batch_Id,  
  
            SUM(ISNULL(D.Despatch_Qty, 0)) AS ProductionQty,  
  
            SUM(ISNULL(D.Loss_Qty, 0)) AS WastageQty  
  
        FROM Despatch D  
        INNER JOIN BatchData B  
            ON B.Production_Batch_Id = D.Production_Batch_Id  
        WHERE D.DeletedAt IS NULL  
        GROUP BY  
            D.Production_Batch_Id  
    ),  
  
    -- Other production costs  
    CostData AS  
    (  
        SELECT  
            PC.Production_Batch_Id,  
  
            SUM(ISNULL(PC.Total_Expenses, 0)) AS OtherCost  
  
        FROM Production_Cost_Master PC  
        INNER JOIN BatchData B  
            ON B.Production_Batch_Id = PC.Production_Batch_Id  
        WHERE PC.DeletedAt IS NULL  
        GROUP BY  
            PC.Production_Batch_Id  
    ),  
  
    -- Sales  
    SalesData AS  
    (  
        SELECT  
            D.Production_Batch_Id,  
  
            SUM(ISNULL(BD.Bill_Amount, 0)) AS SalesAmount  
  
        FROM Despatch D  
  
        INNER JOIN BatchData B  
            ON B.Production_Batch_Id = D.Production_Batch_Id  
  
        INNER JOIN Billing_Details BD  
            ON BD.Production_Despatch_Id = D.Production_Despatch_Id  
           AND BD.DeletedAt IS NULL  
  
        INNER JOIN Billing_Master BM  
            ON BM.Billing_Id = BD.Billing_Id  
           AND BM.DeletedAt IS NULL  
  
        WHERE D.DeletedAt IS NULL  
  
        GROUP BY  
            D.Production_Batch_Id  
    ),  
  
    ReportData AS  
    (  
        SELECT  
            B.BatchId,  
  
            B.Start_Date AS [Date],  
  
            B.Production_Batch_Number AS BatchNo,  
  
            ISNULL(RQ.RawMaterialQty, 0) AS RawMaterialQty,  
  
            ISNULL(PD.ProductionQty, 0) AS ProductionQty,  
  
            ISNULL(PD.WastageQty, 0) AS WastageQty,  
  
            ISNULL(MC.MaterialCost, 0) AS MaterialCost,  
  
            ISNULL(CD.OtherCost, 0) AS OtherCost,  
  
            ISNULL(MC.MaterialCost, 0)  
                + ISNULL(CD.OtherCost, 0) AS TotalCost,  
  
            ISNULL(SD.SalesAmount, 0) AS SalesAmount  
  
        FROM BatchData B  
  
        LEFT JOIN RawMaterialQtyData RQ  
            ON RQ.Production_Batch_Id = B.Production_Batch_Id  
  
        LEFT JOIN MaterialCostData MC  
            ON MC.Production_Batch_Id = B.Production_Batch_Id  
  
        LEFT JOIN ProductionData PD  
            ON PD.Production_Batch_Id = B.Production_Batch_Id  
  
        LEFT JOIN CostData CD  
            ON CD.Production_Batch_Id = B.Production_Batch_Id  
  
        LEFT JOIN SalesData SD  
            ON SD.Production_Batch_Id = B.Production_Batch_Id  
    ),  
  
    CalculatedData AS  
    (  
        SELECT  
            BatchId,  
            [Date],  
            BatchNo,  
            RawMaterialQty,  
            ProductionQty,  
            WastageQty,  
  
            CASE  
                WHEN RawMaterialQty = 0 THEN 0  
                ELSE ProductionQty / RawMaterialQty * 100  
            END AS YieldPercentage,  
  
            MaterialCost,  
            OtherCost,  
            TotalCost,  
  
            CASE  
                WHEN ProductionQty = 0 THEN 0  
                ELSE TotalCost / ProductionQty  
            END AS CostPerKg,  
  
            SalesAmount,  
  
            SalesAmount - TotalCost AS ProfitLoss,  
  
            CASE  
                WHEN TotalCost = 0 THEN 0  
                ELSE  
                    (SalesAmount - TotalCost) / TotalCost * 100  
            END AS ProfitPercentage  
  
        FROM ReportData  
    ),  
  
    FinalData AS  
    (  
        -- Detail rows  
        SELECT  
            0 AS SortOrder,  
            BatchId,  
            [Date],  
            BatchNo,  
            RawMaterialQty,  
            ProductionQty,  
            WastageQty,  
            YieldPercentage,  
            MaterialCost,  
            OtherCost,  
            TotalCost,  
            CostPerKg,  
            SalesAmount,  
            ProfitLoss,  
            ProfitPercentage,  
            CAST(0 AS BIT) AS IsTotal  
        FROM CalculatedData  
  
        UNION ALL  
  
        -- Total row  
        SELECT  
            1 AS SortOrder,  
            NULL AS BatchId,  
            NULL AS [Date],  
            'TOTAL' AS BatchNo,  
  
            SUM(RawMaterialQty) AS RawMaterialQty,  
            SUM(ProductionQty) AS ProductionQty,  
            SUM(WastageQty) AS WastageQty,  
  
            CASE  
                WHEN SUM(RawMaterialQty) = 0 THEN 0  
                ELSE  
                    SUM(ProductionQty)  
                    / SUM(RawMaterialQty) * 100  
            END AS YieldPercentage,  
  
            SUM(MaterialCost) AS MaterialCost,  
            SUM(OtherCost) AS OtherCost,  
            SUM(TotalCost) AS TotalCost,  
  
            CASE  
                WHEN SUM(ProductionQty) = 0 THEN 0  
                ELSE  
                    SUM(TotalCost)  
                    / SUM(ProductionQty)  
            END AS CostPerKg,  
  
            SUM(SalesAmount) AS SalesAmount,  
  
            SUM(SalesAmount) - SUM(TotalCost) AS ProfitLoss,  
  
            CASE  
                WHEN SUM(TotalCost) = 0 THEN 0  
                ELSE  
                    (SUM(SalesAmount) - SUM(TotalCost))  
                    / SUM(TotalCost) * 100  
            END AS ProfitPercentage,  
  
            CAST(1 AS BIT) AS IsTotal  
  
        FROM CalculatedData  
    )  
  
    SELECT  
        [Date],  
        BatchNo,  
        RawMaterialQty,  
        ProductionQty,  
        WastageQty,  
        YieldPercentage,  
        MaterialCost,  
        OtherCost,  
        TotalCost,  
        CostPerKg,  
        SalesAmount,  
        ProfitLoss,  
        ProfitPercentage,  
        IsTotal  
  
    FROM FinalData  
  
    ORDER BY  
        SortOrder,  
        [Date],  
        BatchId;  
  
END;  


GO

create   PROCEDURE dbo.GetPurchaseSupplierOutstandingReport  
(  
    @SearchText NVARCHAR(100) = NULL,  
    @FromDate   DATETIME2 = NULL,  
    @ToDate     DATETIME2 = NULL  
)  
AS  
BEGIN  
    SET NOCOUNT ON;  
  
 select pv.Payment_Voucher_Number,pv.Payment_Voucher_Date,pv.Reference_Number as BillNo,sp.Supplier_Name,pv.Total_Amount as AdvAmount
,jm.Total_Amount as JobRecAmount,isnull(Pv.Total_Amount,0)-isnull(Jm.Total_Amount,0) as BalAmount
,case when jm.payment_status=1 then 'Pending' when jm.payment_status=2 then 'Paid' else 'Partially Paid' end as Payment_status
from payment_voucher pv
left join supplier sp on sp.id=pv.id
left join job_receipt_master jm on jm.supplier_id=sp.supplier_id
where supplier_type='P'
 
  AND pv.Payment_Voucher_Number IS NOT NULL
      AND pv.DeletedAt IS NULL
      AND (
          NULLIF(LTRIM(RTRIM(@SearchText)), '') IS NULL
          OR pv.Payment_Voucher_Number LIKE '%' + LTRIM(RTRIM(@SearchText)) + '%'
          OR pv.Reference_Number LIKE '%' + LTRIM(RTRIM(@SearchText)) + '%'
          OR sp.Supplier_Name LIKE '%' + LTRIM(RTRIM(@SearchText)) + '%'
      )
      AND (@FromDate IS NULL OR pv.Payment_Voucher_Date >= @FromDate)
      AND (@ToDate IS NULL OR pv.Payment_Voucher_Date < DATEADD(DAY, 1, CAST(@ToDate AS DATE)))
order by pv.id desc 


END;
